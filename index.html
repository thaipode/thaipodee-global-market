// ============================================================== // ThaiPodee v1132 — FIX เลือกไฟล์แล้วไม่หาย // อัปเดต: 30 กันยายน 2569 20:45 น. // ============================================================== const fileInput = document.getElementById('file-input'); const uploadTrigger = document.getElementById('upload-trigger'); // กดปุ่ม + → เปิดเลือกไฟล์ uploadTrigger.addEventListener('click', function(e) { e.preventDefault(); e.stopPropagation(); fileInput.click(); }); // เลือกไฟล์แล้ว → ประมวลผลอย่างปลอดภัย fileInput.addEventListener('change', async function(e) { try { const file = e.target.files[0]; if (!file) { console.log('ไม่ได้เลือกไฟล์'); return; } console.log('เลือกไฟล์:', file.name, 'ขนาด:', (file.size / 1024 / 1024).toFixed(2), 'MB'); // ตรวจสอบประเภทไฟล์ if (!file.type.startsWith('video/')) { alert('กรุณาเลือกไฟล์วิดีโอเท่านั้น'); fileInput.value = ''; return; } // แสดงหน้าจอโหลดอย่างเดียว ไม่ให้ขาว showUploadingScreen(file); } catch (err) { console.error('เกิดข้อผิดพลาด:', err); alert('เกิดข้อผิดพลาด: ' + err.message); fileInput.value = ''; } }); // แสดงหน้าจออัปโหลด — ไม่ให้ขาว function showUploadingScreen(file) { // ล้างหน้าจอเก่า const body = document.body; // สร้างหน้าจอแสดงสถานะ body.innerHTML = `

กำลังเตรียมวิดีโอ...

ชื่อไฟล์: ${file.name}

ขนาด: ${(file.size / 1024 / 1024).toFixed(2)} MB

`; // จำลองความคืบหน้า let progress = 0; const bar = document.getElementById('progress-bar'); const interval = setInterval(() => { progress += Math.random() * 15; if (progress >= 100) { progress = 100; clearInterval(interval); // เริ่มอัปโหลดจริง doUpload(file); } if (bar) bar.style.width = progress + '%'; }, 300); } // ย้อนกลับไปหน้าแรก function goBack() { window.location.reload(); } // อัปโหลดไป Firebase Storage async function doUpload(file) { try { console.log('เริ่มอัปโหลด...'); // สร้างชื่อไฟล์ไม่ซ้ำกัน const fileName = Date.now() + '_' + file.name; const filePath = 'videos/' + fileName; // อ้างอิง Firebase Storage const storageRef = firebase.storage().ref(filePath); // อัปโหลดไฟล์ const snapshot = await storageRef.put(file); console.log('อัปโหลดสำเร็จ:', snapshot); // รับลิงก์สาธารณะ const downloadURL = await snapshot.ref.getDownloadURL(); console.log('ลิงก์วิดีโอ:', downloadURL); // บันทึกข้อมูลลง Firestore await saveVideoData(file, downloadURL); // แจ้งสำเร็จ showSuccessScreen(downloadURL); } catch (err) { console.error('อัปโหลดล้มเหลว:', err); alert('อัปโหลดไม่สำเร็จ: ' + err.message); goBack(); } } // บันทึกข้อมูลวิดีโอลงฐานข้อมูล async function saveVideoData(file, videoURL) { await firebase.firestore().collection('videos').add({ fileName: file.name, videoURL: videoURL, fileSize: file.size, fileType: file.type, uploadedAt: firebase.firestore.FieldValue.serverTimestamp(), likes: 0, comments: 0, shares: 0 }); } // แสดงหน้าจอสำเร็จ function showSuccessScreen(videoURL) { document.body.innerHTML = `

✅ อัปโหลดสำเร็จ!

วิดีโอของคุณจะปรากฏในหน้าแรกเร็ว ๆ นี้

`; }
