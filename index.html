// เพิ่มฟังก์ชันนี้ก่อน renderPosts async function rebuildOldThumbs() { posts = JSON.parse(localStorage.getItem('thaiPodeePosts') || '[]'); let changed = false; for (let i = 0; i < posts.length; i++) { const p = posts[i]; if (!p.thumbnail && p.url) { try { const video = document.createElement('video'); video.crossOrigin = 'anonymous'; video.src = p.url; video.preload = 'metadata'; video.currentTime = 1; await new Promise((resolve) => { video.onseeked = () => { const c = document.createElement('canvas'); c.width = video.videoWidth / 2 || 160; c.height = video.videoHeight / 2 || 213; c.getContext('2d').drawImage(video, 0, 0, c.width, c.height); p.thumbnail = c.toDataURL('image/jpeg', 0.5); changed = true; resolve(); }; video.onerror = () => resolve(); setTimeout(resolve, 3000); }); } catch (e) { /* ข้ามถ้าไม่ได้ */ } } } if (changed) { localStorage.setItem('thaiPodeePosts', JSON.stringify(posts)); renderPosts(); } } // เรียกใช้ท้ายสคริปต์ rebuildOldThumbs();
