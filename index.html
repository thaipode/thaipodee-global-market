กำลังเตรียมวิดีโอ...

ชื่อไฟล์: ${file.name}

ขนาด: ${fileSizeMB} MB

`; let progress = 0; const bar = document.getElementById('progress-bar'); const interval = setInterval(() => { progress += Math.random() * 12; if (progress >= 100) { progress = 100; clearInterval(interval); doUpload(file); } if (bar) bar.style.width = progress + '%'; }, 300); } function goBack() { window.location.reload(); } async function doUpload(file) { try { console.log('เริ่มอัปโหลด...'); if (!window.firebase) { showMessage('ไม่พบระบบ Firebase — กลับไปหน้าแรก', false); return; } const fileName = Date.now() + '_' + file.name; const filePath = 'videos/' + fileName; const storageRef = firebase.storage().ref(filePath); const snapshot = await storageRef.put(file); const downloadURL = await snapshot.ref.getDownloadURL(); await firebase.firestore().collection('videos').add({ fileName: file.name, videoURL: downloadURL, fileSize: file.size, fileType: file.type, uploadedAt: firebase.firestore.FieldValue.serverTimestamp(), likes: 0, comments: 0, shares: 0 }); showSuccessScreen(); } catch (err) { console.error('อัปโหลดล้มเหลว:', err); showMessage('อัปโหลดไม่สำเร็จ: ' + err.message, false); } } function showSuccessScreen() { document.body.innerHTML = `

✅ อัปโหลดสำเร็จ!

วิดีโอจะแสดงในหน้าแรกทันที

`; } function showMessage(text, isSuccess) { document.body.innerHTML = `

${isSuccess ? '✅ สำเร็จ' : '❌ ผิดพลาด'}

${text}

`; }
