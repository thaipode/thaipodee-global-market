import express from 'express'; import cors from 'cors'; const app = express(); app.use(cors()); app.use(express.json({limit: '25mb'})); app.use(express.static('.')); const REPO='thaipodee/thaipodee-global-market'; const BRANCH='main'; const getToken=()=>{ const t=process.env.GITHUB_TOKEN; if(!t) throw new Error('ยังไม่ได้ตั้ง GITHUB_TOKEN'); return t.trim(); }; // โหลดรายการ app.get('/api/videos', async (req,res)=>{ try{ const token=getToken(); const r=await fetch(`https://api.github.com/repos/${REPO}/contents/data/videos.json?ref=${BRANCH}`,{headers:{'Authorization':`token ${token}`}}); if(r.status===404) return res.json([]); if(!r.ok) return res.status(r.status).send(await r.text()); const d=await r.json(); const videos=JSON.parse(Buffer.from(d.content,'base64').toString('utf-8')); res.json(videos); }catch(e){ res.status(500).send(e.message); } }); // Proxy วิดีโอ - ทำให้ Private ก็เล่นได้ app.get('/api/file', async (req,res)=>{ try{ const token=getToken(); const path=req.query.path; if(!path) return res.status(400).send('no path'); const r=await fetch(`https://api.github.com/repos/${REPO}/contents/${path}?ref=${BRANCH}`,{headers:{'Authorization':`token ${token}`,'Accept':'application/vnd.github.v3.raw'}}); if(!r.ok) return res.status(r.status).send(await r.text()); res.setHeader('Content-Type', r.headers.get('content-type')||'video/mp4'); res.setHeader('Cache-Control','public, max-age=3600'); const buf=Buffer.from(await r.arrayBuffer()); res.send(buf); }catch(e){ res.status(500).send(e.message); } }); // อัปโหลด app.post('/api/upload', async (req,res)=>{ try{ const token=getToken(); const {fileName,content,name,sizeKB}=req.body; const up=await fetch(`https://api.github.com/repos/${REPO}/contents/${fileName}`,{ method:'PUT', headers:{'Authorization':`token ${token}`,'Content-Type':'application/json'}, body: JSON.stringify({message:`upload ${name}`,content,branch:BRANCH}) }); if(!up.ok) return res.status(up.status).send(await up.text()); let videos=[],sha=null; try{ const rr=await fetch(`https://api.github.com/repos/${REPO}/contents/data/videos.json?ref=${BRANCH}`,{headers:{'Authorization':`token ${token}`}}); if(rr.ok){ const d=await rr.json(); sha=d.sha; videos=JSON.parse(Buffer.from(d.content,'base64').toString('utf-8')); } }catch(e){} videos.push({name,path:fileName,sizeKB,time:new Date().toLocaleString('th-TH'),timestamp:Date.now()}); if(videos.length>50) videos=videos.slice(-50); const newContent=Buffer.from(JSON.stringify(videos,null,2),'utf-8').toString('base64'); await fetch(`https://api.github.com/repos/${REPO}/contents/data/videos.json`,{ method:'PUT', headers:{'Authorization':`token ${token}`,'Content-Type':'application/json'}, body: JSON.stringify({message:`update videos.json`,content:newContent,branch:BRANCH,...(sha?{sha}:{})}) }); res.json({ok:true}); }catch(e){ res.status(500).send(e.message); } }); const PORT=process.env.PORT||10000; app.listen(PORT,()=>console.log('running '+PORT));
