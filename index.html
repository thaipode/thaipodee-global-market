// ===================================================== // ===== แก้หน้าเริ่มต้น — ทำงานแน่นอน เปลี่ยนหน้าเอง ===== // ===================================================== const splash = document.getElementById('splashScreen'); const app = document.getElementById('app'); function goToMain() { console.log('เข้าหน้าหลัก...'); if (splash) splash.classList.add('hidden'); if (app) app.classList.add('visible'); // โหลดข้อมูลทันที if (typeof loadUserData === 'function') loadUserData(); if (typeof loadSavedVideos === 'function') loadSavedVideos(); } // เปลี่ยนหน้าเอง — 3 ช่องทางกันพลาด window.addEventListener('load', () => { setTimeout(goToMain, 1200); // 1.2 วินาที ก็เข้าเอง }); document.addEventListener('DOMContentLoaded', () => { setTimeout(goToMain, 1500); // สำรองอันที่ 1 }); setTimeout(goToMain, 2500); // สำรองอันที่ 2 — ถ้าอันอื่นไม่ทำ อันนี้ทำแน่ // กดที่หน้าจอก็เข้าได้ทันที if (splash) { splash.addEventListener('click', goToMain); }
